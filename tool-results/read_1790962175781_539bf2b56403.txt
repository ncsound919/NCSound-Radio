    67→function hashString(s: string): number {
    68→  let h = 5381
    69→  for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0
    70→  return h
    71→}

    73→/** Deterministic art hue from the rights ID — amber/red family only (20..45). */
    74→function hueForRightsId(rightsId: string): number {
    75→  return 20 + (hashString(rightsId) % 26)
    76→}

    78→function fmtTime(sec: number): string {
    79→  const s = Math.max(0, Math.round(sec))
    80→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`

    83→function timeAgo(playedAtMs: number, serverMs: number): string {
    84→  const min = Math.floor(Math.max(0, serverMs - playedAtMs) / 60000)
    85→  if (min < 1) return 'just now'
    86→  if (min < 60) return `${min} min ago`
    87→  const hrs = Math.floor(min / 60)
    88→  if (hrs < 24) return `${hrs} hr${hrs === 1 ? '' : 's'} ago`
    89→  return `${Math.floor(hrs / 24)} d ago`
    90→}

    92→/** One-second local ticker for smooth progress interpolation between polls. */
    93→function useNowTicker(intervalMs = 1000): number {
    94→  const [now, setNow] = useState(() => Date.now())
    95→  useEffect(() => {
    96→    const id = setInterval(() => setNow(Date.now()), intervalMs)
    97→    return () => clearInterval(id)
    98→  }, [intervalMs])
    99→  return now
   100→}

   102→/** Interpolated on-air progress (0..1) between nowplaying polls. */
   103→function useInterpolatedProgress(data: NowPlayingResponse | null, lastFetch: number): number {
   104→  const now = useNowTicker()
   105→  const duration = data?.current.duration ?? 0
   106→  if (!data || lastFetch <= 0 || duration <= 0) return 0
   107→  return Math.min(1, Math.max(0, data.current.progress + (now - lastFetch) / 1000 / duration))
   108→}

   110→/** Tiny one-shot JSON fetcher with manual retry. */
   111→function useJson<T>(url: string, extraKey?: string) {
   112→  const [data, setData] = useState<T | null>(null)
   113→  const [error, setError] = useState(false)
   114→  const [attempt, setAttempt] = useState(0)
   115→
   116→  useEffect(() => {
   117→    let cancelled = false
   118→    const ctrl = new AbortController()
   119→    const timeout = setTimeout(() => ctrl.abort(), 12000)
   120→    fetch(url, { cache: 'no-store', signal: ctrl.signal })
   121→      .then(async (res) => {
   122→        if (!res.ok) throw new Error(`HTTP ${res.status}`)
   123→        return (await res.json()) as T
   124→      })
   125→      .then((json) => {
   126→        if (!cancelled) {
   127→          setData(json)
   128→          setError(false)
   129→        }
   130→      })
   131→      .catch(() => {
   132→        if (!cancelled) setError(true)
   133→      })
   134→      .finally(() => clearTimeout(timeout))
   135→    return () => {
   136→      cancelled = true
   137→      ctrl.abort()
   138→      clearTimeout(timeout)
   139→    }
   140→  }, [url, extraKey, attempt])
   141→
   142→  const retry = useCallback(() => setAttempt((a) => a + 1), [])
   143→  return { data, error, retry }
   144→}

   575→  const daypartNote = data?.daypart
   576→    ? data.daypart.clean
   577→      ? 'Clean Daypart — explicit lyrics held until 7:00 PM ET'

   579→    : 'Deterministic AutoDJ rotation — rights gate enforced'

   583→      className={cn(
   584→        'bg-card/70 transition-shadow duration-500',

   845→function UpNextCard({
   846→  data,
   847→  onSelect,
   848→}: {
   849→  data: NowPlayingResponse | null
   850→  onSelect: (entry: QueueEntry) => void
   851→}) {
   852→  const duration = data?.current.duration ?? 0
   853→  const remaining = data?.current.remaining ?? 0
   854→  const toNextPct = duration > 0 ? Math.min(100, Math.max(0, ((duration - remaining) / duration) * 100)) : 0
   855→  const cueing = remaining <= 0
   856→
   857→  return (
   858→    <Card className="border-border/60 bg-card/70">
   859→      <CardHeader>
   860→        <CardTitle className="flex items-center gap-2 text-base font-semibold">
   861→          <ListMusic className="h-4 w-4 text-primary" aria-hidden="true" />
   862→          Up Next
   863→        </CardTitle>
   864→        <CardDescription>AutoDJ program clock — music, IDs, ad breaks</CardDescription>
   865→      </CardHeader>
   866→      <CardContent className="space-y-3">
   867→        {/* countdown to the next element — fills as the current one plays out */}
   868→        <div className="rounded-md border border-border/50 bg-background/40 px-3 py-2">
   869→          <div className="flex items-center justify-between text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   870→            <span>{cueing ? 'Cueing up next…' : 'Next spin in'}</span>
   871→            <span
   872→              className={cn(
   873→                'font-mono',
   874→                cueing ? 'animate-pulse text-red-400' : 'text-primary',
   875→              )}
   876→            >
   877→              ~{fmtTime(remaining)}
   878→            </span>
   879→          </div>
   880→          <Progress
   881→            value={toNextPct}
   882→            className={cn('mt-1.5 h-1', cueing && 'animate-pulse')}
   883→            aria-label="Time until the next element"
   884→          />
   885→        </div>
   886→
   887→        {!data ? (
   888→          <div className="space-y-2">
   889→            {[0, 1, 2].map((i) => (
   890→              <Skeleton key={i} className="h-12 w-full rounded-md" />
   891→            ))}
   892→          </div>
   893→        ) : data.next.length === 0 ? (
   894→          <p className="py-6 text-center text-sm text-muted-foreground">Queue is empty.</p>
   895→        ) : (
   896→          <div className="space-y-1">
   897→            {data.next.map((t, i) => {
   898→              const heat = data.heat?.[t.id] ?? 0
   899→              const requesters = data.requestedBy?.[t.id] ?? []
   900→              const isId = t.elementKind === 'STATION_ID'
   901→              const isAd = t.elementKind === 'AD_SPOT'
   902→              const isTalk = t.elementKind === 'TALK'
   903→              const isPseudo = isAd || t.id.startsWith('adspot-') || t.id.startsWith('fallback-')
   904→              const RowTag = isPseudo ? 'div' : 'button'
   905→              return (
   906→                <RowTag
   907→                  key={`${t.id}-${i}`}
   908→                  {...(isPseudo
   909→                    ? {}
   910→                    : {
   911→                        type: 'button' as const,
   912→                        onClick: () => onSelect(t),
   913→                        'aria-label': `Details for ${t.title}`,
   914→                      })}
   915→                  className={cn(
   916→                    'flex w-full items-start gap-3 rounded-md border border-transparent px-2 py-2 text-left transition-colors hover:border-border hover:bg-accent/40',
   917→                    isAd && 'bg-red-500/[0.04]',
   918→                    (isId || isTalk) && 'border-dashed border-border/60 bg-background/30',
   919→                  )}
   920→                >
   921→                  <span
   922→                    className={cn(
   923→                      'flex h-6 w-6 shrink-0 items-center justify-center rounded border font-mono text-[10px]',
   924→                      isAd
   925→                        ? 'border-red-500/40 bg-red-500/10 text-red-400'
   926→                        : isId
   927→                          ? 'border-primary/40 bg-primary/10 text-primary'
   928→                          : isTalk
   929→                            ? 'border-amber-500/40 bg-amber-500/10 text-amber-500'
   930→                            : 'border-border bg-background/60 text-muted-foreground',
   931→                    )}
   932→                  >
   933→                    {isAd ? (
   934→                      <Megaphone className="h-3 w-3" aria-hidden="true" />
   935→                    ) : isId ? (
   936→                      <Mic className="h-3 w-3" aria-hidden="true" />
   937→                    ) : isTalk ? (
   938→                      <MessageSquareQuote className="h-3 w-3" aria-hidden="true" />
   939→                    ) : (
   940→                      i + 1
   941→                    )}
   942→                  </span>
   943→                  <div className="min-w-0 flex-1">
   944→                    <p
   945→                      className={cn(
   946→                        'truncate text-sm font-medium',
   947→                        isAd && 'text-red-300',

   971→                  <Badge

   997→    </Card>
