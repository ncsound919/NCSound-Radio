    56→function hashString(s: string): number {
    57→  let h = 5381
    58→  for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0
    59→  return h
    60→}

    62→/** Deterministic art hue from the rights ID — amber/red family only (20..45). */
    63→function hueForRightsId(rightsId: string): number {
    64→  return 20 + (hashString(rightsId) % 26)
    65→}

    67→function fmtTime(sec: number): string {
    68→  const s = Math.max(0, Math.round(sec))
    69→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`
    70→}

    72→function timeAgo(playedAtMs: number, serverMs: number): string {
    73→  const min = Math.floor(Math.max(0, serverMs - playedAtMs) / 60000)
    74→  if (min < 1) return 'just now'
    75→  if (min < 60) return `${min} min ago`
    76→  const hrs = Math.floor(min / 60)
    77→  if (hrs < 24) return `${hrs} hr${hrs === 1 ? '' : 's'} ago`
    78→  return `${Math.floor(hrs / 24)} d ago`
    79→}

    81→/** One-second local ticker for smooth progress interpolation between polls. */
    82→function useNowTicker(intervalMs = 1000): number {
    83→  const [now, setNow] = useState(() => Date.now())
    84→  useEffect(() => {
    85→    const id = setInterval(() => setNow(Date.now()), intervalMs)
    86→    return () => clearInterval(id)
    87→  }, [intervalMs])
    88→  return now
    89→}

    91→/** Interpolated on-air progress (0..1) between nowplaying polls. */
    92→function useInterpolatedProgress(data: NowPlayingResponse | null, lastFetch: number): number {
    93→  const now = useNowTicker()
    94→  const duration = data?.current.duration ?? 0
    95→  if (!data || lastFetch <= 0 || duration <= 0) return 0
    96→  return Math.min(1, Math.max(0, data.current.progress + (now - lastFetch) / 1000 / duration))
    97→}

    99→/** Tiny one-shot JSON fetcher with manual retry. */
   100→function useJson<T>(url: string, extraKey?: string) {
   101→  const [data, setData] = useState<T | null>(null)
   102→  const [error, setError] = useState(false)
   103→  const [attempt, setAttempt] = useState(0)
   104→
   105→  useEffect(() => {
   106→    let cancelled = false
   107→    const ctrl = new AbortController()
   108→    const timeout = setTimeout(() => ctrl.abort(), 12000)
   109→    fetch(url, { cache: 'no-store', signal: ctrl.signal })
   110→      .then(async (res) => {
   111→        if (!res.ok) throw new Error(`HTTP ${res.status}`)
   112→        return (await res.json()) as T
   113→      })
   114→      .then((json) => {
   115→        if (!cancelled) {
   116→          setData(json)
   117→          setError(false)
   118→        }
   119→      })
   120→      .catch(() => {
   121→        if (!cancelled) setError(true)
   122→      })
   123→      .finally(() => clearTimeout(timeout))
   124→    return () => {
   125→      cancelled = true
   126→      ctrl.abort()
   127→      clearTimeout(timeout)
   128→    }
   129→  }, [url, extraKey, attempt])
   130→
   131→  const retry = useCallback(() => setAttempt((a) => a + 1), [])
   132→  return { data, error, retry }
   133→}
