    67→    67→function hashString(s: string): number {
    68→    68→  let h = 5381
    69→    69→  for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0
    70→    70→  return h
    71→    71→}

    83→    83→function timeAgo(playedAtMs: number, serverMs: number): string {
    84→    84→  const min = Math.floor(Math.max(0, serverMs - playedAtMs) / 60000)
    85→    85→  if (min < 1) return 'just now'
    86→    86→  if (min < 60) return `${min} min ago`
    87→    87→  const hrs = Math.floor(min / 60)
    88→    88→  if (hrs < 24) return `${hrs} hr${hrs === 1 ? '' : 's'} ago`
    89→    89→  return `${Math.floor(hrs / 24)} d ago`
    90→    90→}

    92→    92→/** One-second local ticker for smooth progress interpolation between polls. */
    93→    93→function useNowTicker(intervalMs = 1000): number {
    94→    94→  const [now, setNow] = useState(() => Date.now())
    95→    95→  useEffect(() => {
    96→    96→    const id = setInterval(() => setNow(Date.now()), intervalMs)
    97→    97→    return () => clearInterval(id)
    98→    98→  }, [intervalMs])
    99→    99→  return now
   100→   100→}

   102→   102→/** Interpolated on-air progress (0..1) between nowplaying polls. */
   103→   103→function useInterpolatedProgress(data: NowPlayingResponse | null, lastFetch: number): number {
   104→   104→  const now = useNowTicker()
   105→   105→  const duration = data?.current.duration ?? 0
   106→   106→  if (!data || lastFetch <= 0 || duration <= 0) return 0
   107→   107→  return Math.min(1, Math.max(0, data.current.progress + (now - lastFetch) / 1000 / duration))
   108→   108→}

   110→   110→/** Tiny one-shot JSON fetcher with manual retry. */
   111→   111→function useJson<T>(url: string, extraKey?: string) {
   112→   112→  const [data, setData] = useState<T | null>(null)
   113→   113→  const [error, setError] = useState(false)
   114→   114→  const [attempt, setAttempt] = useState(0)
   115→   115→
   116→   116→  useEffect(() => {
   117→   117→    let cancelled = false
   118→   118→    const ctrl = new AbortController()
   119→   119→    const timeout = setTimeout(() => ctrl.abort(), 12000)
   120→   120→    fetch(url, { cache: 'no-store', signal: ctrl.signal })
   121→   121→      .then(async (res) => {
   122→   122→        if (!res.ok) throw new Error(`HTTP ${res.status}`)
   123→   123→        return (await res.json()) as T
   124→   124→      })
   125→   125→      .then((json) => {
   126→   126→        if (!cancelled) {
   127→   127→          setData(json)
   128→   128→          setError(false)
   129→   129→        }
   130→   130→      })
   131→   131→      .catch(() => {
   132→   132→        if (!cancelled) setError(true)
   133→   133→      })
   134→   134→      .finally(() => clearTimeout(timeout))
   135→   135→    return () => {
   136→   136→      cancelled = true
   137→   137→      ctrl.abort()
   138→   138→      clearTimeout(timeout)
   139→   139→    }
   140→   140→  }, [url, extraKey, attempt])
   141→   141→
   142→   142→  const retry = useCallback(() => setAttempt((a) => a + 1), [])
   143→   143→  return { data, error, retry }
   144→   144→}
