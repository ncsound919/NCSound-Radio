    54→function hashString(s: string): number {
    55→  let h = 5381
    56→  for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0
    57→  return h
    58→}
    59→
    60→/** Deterministic art hue from the rights ID — amber/red family only (20..45). */
    61→function hueForRightsId(rightsId: string): number {
    62→  return 20 + (hashString(rightsId) % 26)
    63→}

    65→function fmtTime(sec: number): string {
    66→  const s = Math.max(0, Math.round(sec))
    67→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`
    68→}

    70→function timeAgo(playedAtMs: number, serverMs: number): string {
    71→  const min = Math.floor(Math.max(0, serverMs - playedAtMs) / 60000)
    72→  if (min < 1) return 'just now'
    73→  if (min < 60) return `${min} min ago`
    74→  const hrs = Math.floor(min / 60)
    75→  if (hrs < 24) return `${hrs} hr${hrs === 1 ? '' : 's'} ago`
    76→  return `${Math.floor(hrs / 24)} d ago`
    77→}

    79→/** One-second local ticker for smooth progress interpolation between polls. */
    80→function useNowTicker(intervalMs = 1000): number {
    81→  const [now, setNow] = useState(() => Date.now())
    82→  useEffect(() => {
    83→    const id = setInterval(() => setNow(Date.now()), intervalMs)
    84→    return () => clearInterval(id)
    85→  }, [intervalMs])
    86→  return now
    87→}

    89→/** Interpolated on-air progress (0..1) between nowplaying polls. */
    90→function useInterpolatedProgress(data: NowPlayingResponse | null, lastFetch: number): number {
    91→  const now = useNowTicker()
    92→  const duration = data?.current.duration ?? 0
    93→  if (!data || lastFetch <= 0 || duration <= 0) return 0
    94→  return Math.min(1, Math.max(0, data.current.progress + (now - lastFetch) / 1000 / duration))
    95→}

    97→/** Tiny one-shot JSON fetcher with manual retry. */
    98→function useJson<T>(url: string, extraKey?: string) {
    99→  const [data, setData] = useState<T | null>(null)
   100→  const [error, setError] = useState(false)
   101→  const [attempt, setAttempt] = useState(0)
   102→
   103→  useEffect(() => {
   104→    let cancelled = false
   105→    const ctrl = new AbortController()
   106→    const timeout = setTimeout(() => ctrl.abort(), 12000)
   107→    fetch(url, { cache: 'no-store', signal: ctrl.signal })
   108→      .then(async (res) => {
   109→        if (!res.ok) throw new Error(`HTTP ${res.status}`)
   110→        return (await res.json()) as T
   111→      })
   112→      .then((json) => {
   113→        if (!cancelled) {
   114→          setData(json)
   115→          setError(false)
   116→        }
   117→      })
   118→      .catch(() => {
   119→        if (!cancelled) setError(true)
   120→      })
   121→      .finally(() => clearTimeout(timeout))
   122→    return () => {
   123→      cancelled = true
   124→      ctrl.abort()
   125→      clearTimeout(timeout)
   126→    }
   127→  }, [url, extraKey, attempt])
   128→
   129→  const retry = useCallback(() => setAttempt((a) => a + 1), [])
   130→  return { data, error, retry }
   131→}
