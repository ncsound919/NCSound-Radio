    48→function hashString(s: string): number {
    49→  let h = 5381
    50→  for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0
    51→  return h
    52→}
    53→
    54→/** Deterministic art hue from the rights ID — amber/red family only (20..45). */
    55→function hueForRightsId(rightsId: string): number {
    56→  return 20 + (hashString(rightsId) % 26)
    57→}

    59→function fmtTime(sec: number): string {
    60→  const s = Math.max(0, Math.round(sec))
    61→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`
    62→}

    64→function timeAgo(playedAtMs: number, serverMs: number): string {
    65→  const min = Math.floor(Math.max(0, serverMs - playedAtMs) / 60000)
    66→  if (min < 1) return 'just now'
    67→  if (min < 60) return `${min} min ago`
    68→  const hrs = Math.floor(min / 60)
    69→  if (hrs < 24) return `${hrs} hr${hrs === 1 ? '' : 's'} ago`
    70→  return `${Math.floor(hrs / 24)} d ago`
    71→}

    73→/** One-second local ticker for smooth progress interpolation between polls. */
    74→function useNowTicker(intervalMs = 1000): number {
    75→  const [now, setNow] = useState(() => Date.now())
    76→  useEffect(() => {
    77→    const id = setInterval(() => setNow(Date.now()), intervalMs)
    78→    return () => clearInterval(id)
    79→  }, [intervalMs])
    80→  return now
    81→}

    83→/** Interpolated on-air progress (0..1) between nowplaying polls. */
    84→function useInterpolatedProgress(data: NowPlayingResponse | null, lastFetch: number): number {
    85→  const now = useNowTicker()
    86→  const duration = data?.current.duration ?? 0
    87→  if (!data || lastFetch <= 0 || duration <= 0) return 0
    88→  return Math.min(1, Math.max(0, data.current.progress + (now - lastFetch) / 1000 / duration))
    89→}

    91→/** Tiny one-shot JSON fetcher with manual retry. */
    92→function useJson<T>(url: string, extraKey?: string) {
    93→  const [data, setData] = useState<T | null>(null)
    94→  const [error, setError] = useState(false)
    95→  const [attempt, setAttempt] = useState(0)
    96→
    97→  useEffect(() => {
    98→    let cancelled = false
    99→    const ctrl = new AbortController()
   100→    const timeout = setTimeout(() => ctrl.abort(), 12000)
   101→    fetch(url, { cache: 'no-store', signal: ctrl.signal })
   102→      .then(async (res) => {
   103→        if (!res.ok) throw new Error(`HTTP ${res.status}`)
   104→        return (await res.json()) as T
   105→      })
   106→      .then((json) => {
   107→        if (!cancelled) {
   108→          setData(json)
   109→          setError(false)
   110→        }
   111→      })
   112→      .catch(() => {
   113→        if (!cancelled) setError(true)
   114→      })
   115→      .finally(() => clearTimeout(timeout))
   116→    return () => {
   117→      cancelled = true
   118→      ctrl.abort()
   119→      clearTimeout(timeout)
   120→    }
   121→  }, [url, extraKey, attempt])
   122→
   123→  const retry = useCallback(() => setAttempt((a) => a + 1), [])
   124→  return { data, error, retry }
   125→}
